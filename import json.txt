import json
import re
import os

def load_thirukkural(filepath):
    with open(filepath, "r", encoding="utf-8") as f:
        data = json.load(f)
    results = []
    for item in data["kural"]:
        results.append({
            "id": f"thirukkural_{str(item['Number']).zfill(4)}",
            "source": "thirukkural",
            "number": item["Number"],
            "original_tamil": item["Line1"] + " " + item["Line2"],
            "line1": item.get("Line1", ""),
            "line2": item.get("Line2", ""),
            "transliteration": item.get("transliteration1", "") + " " + item.get("transliteration2", ""),
            "translation_en": item.get("Translation") or item.get("explanation") or "",
            "translation_ta": item.get("mv", ""),
            "explanation": item.get("explanation", ""),
            "couplet": item.get("couplet", ""),
            "theme": "",
            "chapter": "",
            "poet": "திருவள்ளுவர்",
            "period": "sangam"
        })
    print(f"  Thirukkural: {len(results)} kurals loaded")
    return results

def load_purananuru(filepath):
    with open(filepath, "r", encoding="utf-8") as f:
        content = f.read()
    blocks = re.split(r'\n(?=\d+\.)', content)
    results = []
    for block in blocks:
        block = block.strip()
        if not block:
            continue
        lines = block.split("\n")
        first_line = lines[0].strip()
        num_match = re.match(r'^(\d+)\.\s*(.*)', first_line)
        if not num_match:
            continue
        poem_number = int(num_match.group(1))
        title = num_match.group(2).strip()
        poet = ""
        subject = ""
        tinai = ""
        thurai = ""
        poem_lines = []
        in_poem = False
        for line in lines[1:]:
            line = line.strip()
            if not line:
                if poet or tinai:
                    in_poem = True
                continue
            if line.startswith("பாடியவர்"):
                poet = re.sub(r'^பாடியவர்\s*[:]\s*', '', line).strip(" .")
            elif line.startswith("பாடப்பட்டோன்"):
                subject = re.sub(r'^பாடப்பட்டோன்\s*[:]\s*', '', line).strip(" .")
            elif line.startswith("திணை"):
                tinai = re.sub(r'^திணை\s*[:]\s*', '', line).strip(" .")
            elif line.startswith("துறை"):
                thurai = re.sub(r'^துறை\s*[:]\s*', '', line).strip(" .")
            elif line.startswith("சிறப்பு"):
                pass
            else:
                poem_lines.append(line)
        poem_text = "\n".join(poem_lines).strip()
        if not poem_text:
            continue
        results.append({
            "id": f"purananuru_{str(poem_number).zfill(4)}",
            "source": "purananuru",
            "number": poem_number,
            "title": title,
            "original_tamil": poem_text,
            "transliteration": "",
            "translation_en": "",
            "translation_ta": "",
            "explanation": "",
            "poet": poet,
            "subject": subject,
            "tinai": tinai,
            "thurai": thurai,
            "period": "sangam"
        })
    print(f"  Purananuru: {len(results)} poems loaded")
    return results

def load_akananuru(filepath):
    with open(filepath, "r", encoding="utf-8") as f:
        data = json.load(f)
    results = []
    poems = data if isinstance(data, list) else data.get("poems") or data.get("akananuru") or list(data.values())[0]
    for item in poems:
        number = item.get("number") or item.get("Number") or 0
        tamil  = item.get("tamil") or item.get("original_tamil") or item.get("poem") or item.get("text") or ""
        transl = item.get("translation") or item.get("translation_en") or item.get("Translation") or ""
        poet   = item.get("poet") or item.get("author") or ""
        results.append({
            "id": f"akananuru_{str(number).zfill(4)}",
            "source": "akananuru",
            "number": number,
            "original_tamil": tamil,
            "transliteration": item.get("transliteration", ""),
            "translation_en": transl,
            "translation_ta": item.get("translation_ta", ""),
            "explanation": item.get("explanation", ""),
            "poet": poet,
            "tinai": item.get("tinai", ""),
            "thurai": item.get("thurai", ""),
            "period": "sangam"
        })
    print(f"  Akananuru: {len(results)} poems loaded")
    return results

if __name__ == "__main__":
    print("\nTamil AI Corpus Converter")
    print("-" * 40)
    corpus = []
    THIRUKKURAL_PATH = "data/thirukkural.json"
    PURANANURU_PATH  = "data/purananuru_1-100"
    AKANANURU_PATH   = "data/akananuru.json"
    OUTPUT_PATH      = "data/corpus.json"
    if os.path.exists(THIRUKKURAL_PATH):
        corpus += load_thirukkural(THIRUKKURAL_PATH)
    else:
        print(f"  WARNING: Thirukkural not found at {THIRUKKURAL_PATH}")
    if os.path.exists(PURANANURU_PATH):
        corpus += load_purananuru(PURANANURU_PATH)
    else:
        print(f"  WARNING: Purananuru not found at {PURANANURU_PATH}")
    if os.path.exists(AKANANURU_PATH):
        corpus += load_akananuru(AKANANURU_PATH)
    else:
        print(f"  WARNING: Akananuru not found at {AKANANURU_PATH}")
    os.makedirs("data", exist_ok=True)
    with open(OUTPUT_PATH, "w", encoding="utf-8") as f:
        json.dump(corpus, f, ensure_ascii=False, indent=2)
    print(f"\nDone! {len(corpus)} total entries saved to {OUTPUT_PATH}")
    print("-" * 40)
